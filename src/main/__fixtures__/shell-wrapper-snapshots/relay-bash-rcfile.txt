# Why: OMP does not auto-load Shield's managed status extension; wrap only
# interactive launch invocations so subcommands such as `omp config` keep
# their normal argv shape.
__shield_omp_should_skip_extension() {
  case "${1:-}" in
    'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
    '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
  esac
  return 1
}
__shield_omp_cwd_is_usable() {
  local __shield_physical_cwd
  [[ -x . ]] || return 1
  if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
    [[ "${PWD}" -ef . ]]
  else
    # Why compare the path: shell builtins can print a cached path for a deleted cwd.
    __shield_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
    [[ -d "$__shield_physical_cwd" && "$__shield_physical_cwd" -ef . ]]
  fi
}
__shield_omp_invoke() {
  local __shield_use_extension="$1"
  shift
  if [[ $__shield_use_extension -eq 1 && -n "${OSHIELD_OMP_STATUS_EXTENSION:-}" && -f "${OSHIELD_OMP_STATUS_EXTENSION}" ]]; then
    if [[ "${1:-}" == "launch" ]]; then
      shift
      command omp launch --extension "${OSHIELD_OMP_STATUS_EXTENSION}" "$@"
    else
      command omp --extension "${OSHIELD_OMP_STATUS_EXTENSION}" "$@"
    fi
  else
    command omp "$@"
  fi
}
__shield_omp() {
  local __shield_use_extension=1
  __shield_omp_should_skip_extension "${1:-}" && __shield_use_extension=0
  if ! __shield_omp_cwd_is_usable; then
    local __shield_logical_cwd="${PWD:-${OSHIELD_WORKTREE_PATH:-${OSHIELD_ROOT_PATH:-}}}"
    # Why: a restored shell can retain the deleted directory inode after its path is recreated.
    (
      if [[ -z "$__shield_logical_cwd" ]]; then
        printf 'Shield: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
        return 1
      fi
      if ! builtin cd -P -- "$__shield_logical_cwd" 2>/dev/null; then
        printf 'Shield: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__shield_logical_cwd" >&2
        return 1
      fi
      __shield_omp_invoke "$__shield_use_extension" "$@"
    )
  else
    __shield_omp_invoke "$__shield_use_extension" "$@"
  fi
}

  # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
  function omp { __shield_omp "$@"; }

# Why: a typed alias expands inside the shell, after pane launch prep.
# Why unalias inside the substitution: an alias named codex makes command -v
# report the alias text, and the subshell leaves the user's own alias intact.
# Why || : twice — zsh alone aborts inside the substitution, but every shell's
# assignment adopts its exit status, so an absent codex trips set -e in bash too.
__shield_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
if [[ -n "${__shield_codex_binary:-}" && -x "${__shield_codex_binary}" ]]; then
  # Why the function reserved word: it suppresses alias expansion of the name,
  # which otherwise rewrites this header at parse time and aborts the whole file.
  function codex {
    # Why local: zsh's warn_create_global warns for each global a function creates.
    local __shield_codex_arg __shield_codex_isolate="${OSHIELD_CODEX_ISOLATE:-1}"
    if [[ -n "${OSHIELD_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${OSHIELD_CODEX_LAUNCH_PREFLIGHT}" ]]; then
      "${OSHIELD_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
    fi
    for __shield_codex_arg in "$@"; do
      case "$__shield_codex_arg" in agents|queue|--no-daemon|--remote|--remote=*) __shield_codex_isolate=0 ;; esac
    done
    # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
    if [[ "$__shield_codex_isolate" != 0 ]]; then
      case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
    fi
    command codex "$@"
  }
fi

# Why: SSH bash sessions need the same command lifecycle markers as local

__shield_initializing_wrapper=1
__shield_osc133_precmd() {
  local exit_code=$?
  __shield_in_prompt_command=1
  if [[ -n "${__shield_in_command:-}" ]]; then
    printf "\033]133;D;%s\007" "$exit_code"
    unset __shield_in_command
  fi
  printf "\033]133;A\007"
  return "$exit_code"
}
__shield_osc133_prompt_done() {
  unset __shield_in_prompt_command; __shield_adopt_outer_debug_trap
  trap '__shield_osc133_preexec' DEBUG
}
__shield_osc133_preexec() {
  if [[ -n "${__shield_prompt_status_capture_command:-}" && "$BASH_COMMAND" == "$__shield_prompt_status_capture_command" ]]; then
    unset __shield_initial_prompt
    __shield_in_legacy_prompt_wrapper=1
    return 0
  fi
  if [[ -n "${__shield_initializing_wrapper:-}${__shield_in_debug_capture:-}${__shield_initial_prompt:-}${__shield_in_prompt_dispatch:-}${__shield_in_legacy_prompt_wrapper:-}${__shield_in_prompt_command:-}" ]]; then
    [[ -z "${__shield_initializing_wrapper:-}${__shield_in_debug_capture:-}" ]] || return 0
    if [[ -n "${__shield_initial_prompt:-}" && "$BASH_COMMAND" == "__shield_osc133_precmd" ]]; then
      unset __shield_initial_prompt; return 0
    fi
    if [[ -n "${__shield_in_prompt_dispatch:-}" ]]; then
      [[ -n "${__shield_dispatching_user_prompt_command:-}" ]] || return 0
      if [[ "${FUNCNAME[1]:-}" == "__shield_run_prompt_command_array" ]]; then
        case "$BASH_COMMAND" in
          '(( __shield_exit_code == 0 ))'|'__shield_restore_prompt_status "$__shield_exit_code"'|'eval "$__shield_prompt_part"'|'eval "$__shield_final_prompt_command"'|__shield_dispatching_user_prompt_command=*|__shield_osc133_precmd|__shield_osc133_prompt_done|__shield_prompt_mark) return 0 ;;
        esac
      fi
    elif [[ "${FUNCNAME[1]:-}" == "__shield_run_prompt_command_array" || "$BASH_COMMAND" == "__shield_run_prompt_command_array" ]]; then
      return 0
    fi
    [[ -z "${__shield_in_legacy_prompt_wrapper:-}" || -n "${__shield_dispatching_user_prompt_command:-}" ]] || return 0
    if [[ -n "${__shield_in_prompt_command:-}" && "$BASH_COMMAND" == "__shield_in_debug_capture=1" ]]; then
      return 0
    fi
  fi
  case "${FUNCNAME[1]:-}" in __shield_osc133_*|__shield_prompt_mark|__shield_restore_prompt_status) return 0 ;; esac
  case "$BASH_COMMAND" in __shield_osc133_precmd|__shield_osc133_prompt_done|__shield_prompt_mark) return 0 ;; esac
  __shield_run_user_debug_trap
  [[ -z "${__shield_in_prompt_command:-}" ]] || return 0
  [[ -z "${__shield_in_command:-}" ]] || return 0
  printf "\033]133;C\007"
  __shield_in_command=1
}
__shield_normalize_prompt_command_part() {
  local __shield_value="$1" __shield_output_name="$2" __shield_character __shield_chunk
  local __shield_value_length=${#1} __shield_suffix_length=0 __shield_backslash_length=0
  local __shield_output_length __shield_scan_start
  while (( __shield_value_length - __shield_suffix_length >= 1024 )); do
    __shield_scan_start=$(( __shield_value_length - __shield_suffix_length - 1024 ))
    __shield_chunk="${__shield_value:__shield_scan_start:1024}"
    case "$__shield_chunk" in
      *[!$' \t\n;']*) break ;;
      *) __shield_suffix_length=$(( __shield_suffix_length + 1024 )) ;;
    esac
  done
  while (( __shield_suffix_length < __shield_value_length )); do
    __shield_character="${__shield_value: -__shield_suffix_length - 1:1}"
    case "$__shield_character" in
      ' '|$'\t'|$'\n'|';') __shield_suffix_length=$(( __shield_suffix_length + 1 )) ;;
      *) break ;;
    esac
  done
  __shield_output_length=$(( ${#__shield_value} - __shield_suffix_length ))
  while (( __shield_output_length - __shield_backslash_length >= 1024 )); do
    __shield_scan_start=$(( __shield_output_length - __shield_backslash_length - 1024 ))
    __shield_chunk="${__shield_value:__shield_scan_start:1024}"
    case "$__shield_chunk" in
      *[!\\]*) break ;;
      *) __shield_backslash_length=$(( __shield_backslash_length + 1024 )) ;;
    esac
  done
  while (( __shield_backslash_length < __shield_output_length )); do
    __shield_character="${__shield_value:__shield_output_length - __shield_backslash_length - 1:1}"
    [[ "$__shield_character" == '\' ]] || break
    __shield_backslash_length=$(( __shield_backslash_length + 1 ))
  done
  # Preserve the first separator when an odd backslash run escapes it.
  if (( __shield_suffix_length > 0 && __shield_backslash_length % 2 == 1 )); then
    __shield_suffix_length=$(( __shield_suffix_length - 1 ))
    __shield_backslash_length=0
  fi
  __shield_output_length=$(( ${#__shield_value} - __shield_suffix_length ))
  __shield_value="${__shield_value:0:__shield_output_length}"
  # Bash 4.4-5.0 scalar prompt evaluation preserves an odd terminal backslash.
  if (( __shield_suffix_length == 0 && ((BASH_VERSINFO[0] == 4 && BASH_VERSINFO[1] >= 4) || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] == 0)) && __shield_backslash_length % 2 == 1 )); then
    __shield_value="$__shield_value\\"
  fi
  printf -v "$__shield_output_name" '%s' "$__shield_value"
}
__shield_restore_prompt_status() {
  return "$1"
}
__shield_update_user_debug_trap() {
  local __shield_debug_trap_spec="$1" __shield_unchanged_debug_trap_spec="$2"
  local __shield_debug_trap_command
  [[ "$__shield_debug_trap_spec" != "$__shield_unchanged_debug_trap_spec" ]] || return 0
  [[ "$__shield_debug_trap_spec" != "trap -- '__shield_osc133_preexec' DEBUG" ]] || return 0
  if [[ -z "$__shield_debug_trap_spec" ]]; then
    __shield_user_debug_trap=""
    unset __shield_chained_debug_trap
    return 0
  fi
  __shield_debug_trap_command="${__shield_debug_trap_spec#trap -- }"
  __shield_debug_trap_command="${__shield_debug_trap_command% DEBUG}"
  eval "__shield_user_debug_trap=$__shield_debug_trap_command"
  unset __shield_chained_debug_trap
}
__shield_run_user_debug_trap() {
  if [[ -n "${__shield_user_debug_trap:-}" ]]; then
    eval "$__shield_user_debug_trap" || true
  fi
}
__shield_adopt_outer_debug_trap() {
  local __shield_debug_trap_spec="${__shield_outer_debug_trap_spec:-}"
  unset __shield_outer_debug_trap_spec
  __shield_update_user_debug_trap "$__shield_debug_trap_spec" "trap -- '__shield_osc133_preexec' DEBUG"
}
__shield_run_prompt_command_array() {
  local __shield_exit_code="${__shield_prompt_status:-$?}" __shield_prompt_part __shield_prompt_index __shield_user_count
  local __shield_suffix_part
  local __shield_final_prompt_command
  local __shield_in_prompt_dispatch=1 __shield_dispatching_user_prompt_command=""
  unset __shield_prompt_status
  __shield_adopt_outer_debug_trap
  trap '__shield_osc133_preexec' DEBUG
  for __shield_prompt_part in "${__shield_prompt_command_prefix[@]+"${__shield_prompt_command_prefix[@]}"}"; do
    if (( __shield_exit_code == 0 )); then
      eval "$__shield_prompt_part"
    else
      __shield_restore_prompt_status "$__shield_exit_code" || eval "$__shield_prompt_part"
    fi
  done
  __shield_user_count=0
  for __shield_prompt_part in "${__shield_prompt_command_array[@]+"${__shield_prompt_command_array[@]}"}"; do
    __shield_user_count=$(( __shield_user_count + 1 ))
  done
  for (( __shield_prompt_index = 0; __shield_prompt_index + 1 < __shield_user_count; __shield_prompt_index++ )); do
    __shield_prompt_part="${__shield_prompt_command_array[__shield_prompt_index]}"
    __shield_dispatching_user_prompt_command=1
    if (( __shield_exit_code == 0 )); then
      eval "$__shield_prompt_part"
    else
      __shield_restore_prompt_status "$__shield_exit_code" || eval "$__shield_prompt_part"
    fi
    __shield_dispatching_user_prompt_command=""
  done
  if (( __shield_user_count > 0 )); then
    __shield_prompt_part="${__shield_prompt_command_array[__shield_user_count - 1]}"
    # Why: keep the final user hook and Shield suffixes in one status-preserving eval.
    __shield_final_prompt_command='eval "$__shield_prompt_part"'
    for __shield_suffix_part in "${__shield_prompt_command_suffix[@]+"${__shield_prompt_command_suffix[@]}"}"; do
      __shield_final_prompt_command+=$'\n'"$__shield_suffix_part"
    done
    __shield_dispatching_user_prompt_command=1
    if (( __shield_exit_code == 0 )); then
      eval "$__shield_final_prompt_command"
    else
      __shield_restore_prompt_status "$__shield_exit_code" || eval "$__shield_final_prompt_command"
    fi
    __shield_dispatching_user_prompt_command=""
  else
    for __shield_prompt_part in "${__shield_prompt_command_suffix[@]+"${__shield_prompt_command_suffix[@]}"}"; do
      if (( __shield_exit_code == 0 )); then
        eval "$__shield_prompt_part"
      else
        __shield_restore_prompt_status "$__shield_exit_code" || eval "$__shield_prompt_part"
      fi
    done
  fi
  return "$__shield_exit_code"
}
__shield_finish_legacy_prompt_dispatch() {
  local __shield_suffix_part
  if [[ -n "${__shield_in_prompt_command:-}" ]]; then
    for __shield_suffix_part in "${__shield_prompt_command_suffix[@]+"${__shield_prompt_command_suffix[@]}"}"; do
      eval "$__shield_suffix_part"
    done
  fi
  trap '__shield_osc133_preexec' DEBUG
  unset __shield_in_legacy_prompt_wrapper
}
__shield_normalize_prompt_command() {
  [[ -z "${__shield_prompt_command_normalized:-}" ]] || return 0
  local __shield_prompt_part
  local -a __shield_normalized=()
  for __shield_prompt_part in "${PROMPT_COMMAND[@]+"${PROMPT_COMMAND[@]}"}"; do
    __shield_normalize_prompt_command_part "$__shield_prompt_part" __shield_prompt_part
    [[ -n "$__shield_prompt_part" ]] && __shield_normalized+=("$__shield_prompt_part")
  done
  __shield_prompt_command_normalized=1
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND=("${__shield_normalized[@]+"${__shield_normalized[@]}"}")
  else
    __shield_prompt_command_array=("${__shield_normalized[@]+"${__shield_normalized[@]}"}")
    __shield_prompt_command_prefix=()
    __shield_prompt_command_suffix=()
    unset PROMPT_COMMAND
    # Why: PID scope distinguishes legacy prompt dispatch from ordinary user command text.
    __shield_prompt_status_variable="__shield_prompt_status_$$"
    __shield_prompt_status_capture_command="$__shield_prompt_status_variable=\$?"
    __shield_prompt_status_value="\${$__shield_prompt_status_variable}"
    PROMPT_COMMAND="$__shield_prompt_status_capture_command; __shield_prompt_status=$__shield_prompt_status_value"'; __shield_prompt_had_functrace=""; if [[ -o functrace ]]; then __shield_prompt_had_functrace=1; set +T; fi; __shield_outer_debug_trap_spec="$(trap -p DEBUG)"; [[ -z "$__shield_prompt_had_functrace" ]] || set -T; unset __shield_prompt_had_functrace; __shield_run_prompt_command_array; __shield_finish_legacy_prompt_dispatch'
  fi
}
__shield_prepend_prompt_command() {
  local command="$1"
  __shield_normalize_prompt_command
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND=("$command" "${PROMPT_COMMAND[@]+"${PROMPT_COMMAND[@]}"}")
  else
    __shield_prompt_command_prefix=("$command" "${__shield_prompt_command_prefix[@]+"${__shield_prompt_command_prefix[@]}"}")
  fi
}
__shield_append_prompt_command() {
  local command="$1"
  __shield_normalize_prompt_command
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND+=("$command")
  else
    __shield_prompt_command_suffix+=("$command")
  fi
}
