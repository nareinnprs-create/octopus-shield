# Shield relay zsh overlay wrapper
__shield_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Shield's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.shield-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _shield_startup_file
  for _shield_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_shield_startup_file" ]] && return 0
  done
  return 1
}
if __shield_usable_zdotdir "${OSHIELD_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$OSHIELD_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset OSHIELD_ORIG_ZDOTDIR OSHIELD_ZSHENV_SOURCE_DIR
builtin unfunction __shield_usable_zdotdir
builtin typeset -ga _shield_shell_features
_shield_shell_features=(${(s:,:)${OSHIELD_SHELL_FEATURES:-}})
builtin unset OSHIELD_SHELL_FEATURES
# Why OSHIELD_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Shield (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _shield_histfile="${OSHIELD_HISTFILE:-}"
builtin unset OSHIELD_HISTFILE
__shield_has_feature() { (( ${_shield_shell_features[(Ie)$1]} )) }
__shield_has_feature identity && printf "\033]777;shield-shell-start:%s\007" "$$"
__shield_deferred_line_init() {
  builtin emulate -L zsh
  (( ${+functions[__shield_deferred_init]} )) || return 0
  local __shield_direct_line_init=0
  [[ "${widgets[zle-line-init]:-}" == user:__shield_deferred_line_init ]] && __shield_direct_line_init=1
  __shield_deferred_init
  if (( __shield_direct_line_init && ${+widgets[zle-line-init]} )); then
    zle zle-line-init "$@"
  elif [[ "${widgets[zle-line-init]:-}" == user:__shield_prompt_mark ]]; then
    local __shield_prev_line_init_fn=""
    __shield_prompt_mark "$@"
  fi
}
# Why: scheduled callbacks run after user prompt hooks without copying their function metadata.
__shield_deferred_sched_init() {
  local __shield_prompt_status=$?
  builtin emulate -L zsh
  (( ${+functions[__shield_deferred_init]} )) && __shield_deferred_init
  builtin unset __shield_deferred_sched_armed
  builtin unfunction __shield_deferred_sched_init
  return $__shield_prompt_status
}
__shield_arm_deferred_line_init() {
  builtin emulate -L zsh
  if [[ "${widgets[zle-line-init]:-}" != user:__shield_deferred_line_init ]]; then
    if (( ${+widgets[zle-line-init]} )); then
      zle -A zle-line-init __shield_saved_line_init
    fi
    zle -N zle-line-init __shield_deferred_line_init
  fi
  if (( ! $+__shield_deferred_sched_armed )) && builtin zmodload -F zsh/sched b:sched 2>/dev/null; then
    builtin sched +0 __shield_deferred_sched_init && builtin typeset -g __shield_deferred_sched_armed=1
  fi
}
__shield_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_shield_deferred_init_done )) && return 0
  builtin typeset -g _shield_deferred_init_done=1
  builtin typeset -g precmd_functions
  if (( ${+widgets[__shield_saved_line_init]} )); then
    if [[ "${widgets[zle-line-init]:-}" == user:__shield_deferred_line_init ]]; then
      zle -A __shield_saved_line_init zle-line-init
    fi
    zle -D __shield_saved_line_init
  elif [[ "${widgets[zle-line-init]:-}" == user:__shield_deferred_line_init ]]; then
    zle -D zle-line-init
  fi
  precmd_functions=(${precmd_functions:#__shield_deferred_init})
  if [ -n "${OSHIELD_CLI_BIN_DIR:-}" ]; then
    case "${PATH:-}" in
      "$OSHIELD_CLI_BIN_DIR"|"$OSHIELD_CLI_BIN_DIR":*) ;;
      *) export PATH="$OSHIELD_CLI_BIN_DIR${PATH:+:$PATH}" ;;
    esac
  fi
  if __shield_has_feature overlay; then
    # Why: remote startup files can re-export user defaults after relay spawn.
    [[ -n "${OSHIELD_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${OSHIELD_OPENCODE_CONFIG_DIR}"
    if [[ -n "${OSHIELD_DATA_ACCOUNT_DATA_HOME:-}" ]]; then
      export XDG_DATA_HOME="${OSHIELD_DATA_ACCOUNT_DATA_HOME}"
      export XDG_STATE_HOME="${OSHIELD_DATA_ACCOUNT_STATE_HOME}"
      if [[ "${OSHIELD_DATA_ACCOUNT_PROVIDER:-}" == opencode ]]; then
        export OPENCODE_AUTH_CONTENT=""
        export OPENCODE_DB="opencode.db"
      fi
    fi
    [[ -n "${OSHIELD_MIMOCODE_CONFIG_DIR:-}" ]] && export MIMOCODE_CONFIG_DIR="${OSHIELD_MIMOCODE_CONFIG_DIR}"
    [[ -n "${OSHIELD_REMOTE_CLI_BIN_DIR:-}" ]] && case ":$PATH:" in *:"${OSHIELD_REMOTE_CLI_BIN_DIR}":*) ;; *) export PATH="${OSHIELD_REMOTE_CLI_BIN_DIR}:$PATH" ;; esac
    # Why: OMP does not auto-load Shield's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __shield_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __shield_omp_cwd_is_usable() {
      local __shield_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __shield_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__shield_physical_cwd" && "$__shield_physical_cwd" -ef . ]]
      fi
    }
    __shield_omp_invoke() {
      local __shield_use_extension="$1"
      shift
      if [[ $__shield_use_extension -eq 1 && -n "${OSHIELD_OMP_STATUS_EXTENSION:-}" && -f "${OSHIELD_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${OSHIELD_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${OSHIELD_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __shield_omp() {
      local __shield_use_extension=1
      __shield_omp_should_skip_extension "${1:-}" && __shield_use_extension=0
      if ! __shield_omp_cwd_is_usable; then
        local __shield_logical_cwd="${PWD:-${OSHIELD_WORKTREE_PATH:-${OSHIELD_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__shield_logical_cwd" ]]; then
            printf 'Shield: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__shield_logical_cwd" 2>/dev/null; then
            printf 'Shield: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__shield_logical_cwd" >&2
            return 1
          fi
          __shield_omp_invoke "$__shield_use_extension" "$@"
        )
      else
        __shield_omp_invoke "$__shield_use_extension" "$@"
      fi
    }
    if [[ -n "${OSHIELD_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __shield_omp "$@"; }
    fi
  fi
  # Why: a typed alias expands inside the shell, after pane launch prep.
  # Why unalias inside the substitution: an alias named codex makes command -v
  # report the alias text, and the subshell leaves the user's own alias intact.
  # Why || : twice — zsh alone aborts inside the substitution, but every shell's
  # assignment adopts its exit status, so an absent codex trips set -e in bash too.
  __shield_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
  if [[ -n "${__shield_codex_binary:-}" && -x "${__shield_codex_binary}" ]]; then
    # Why the function reserved word: it suppresses alias expansion of the name,
    # which otherwise rewrites this header at parse time and aborts the whole file.
    function codex {
      # Why local: zsh's warn_create_global warns for each global a function creates.
      local __shield_codex_arg __shield_codex_isolate="${OSHIELD_CODEX_ISOLATE:-1}"
      if [[ -n "${OSHIELD_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${OSHIELD_CODEX_LAUNCH_PREFLIGHT}" ]]; then
        "${OSHIELD_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
      fi
      for __shield_codex_arg in "$@"; do
        case "$__shield_codex_arg" in agents|queue|--no-daemon|--remote|--remote=*) __shield_codex_isolate=0 ;; esac
      done
      # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
      if [[ "$__shield_codex_isolate" != 0 ]]; then
        case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
      fi
      command codex "$@"
    }
  fi
  unset __shield_codex_binary
  __shield_claude_binary="$(unalias claude 2>/dev/null || :; command -v claude 2>/dev/null || :)"
  if [[ -n "${OSHIELD_CLAUDE_PROFILE_POINTER:-}" && -n "${__shield_claude_binary:-}" && -x "${__shield_claude_binary}" ]]; then
    function claude {
      local __shield_claude_home __shield_claude_pointer="${OSHIELD_CLAUDE_PROFILE_POINTER:-}"
      # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
      case "$__shield_claude_pointer" in '~/'*) __shield_claude_pointer="${HOME:-}/${__shield_claude_pointer#??}" ;; esac
      __shield_claude_home="$(cat "$__shield_claude_pointer" 2>/dev/null || :)"
      if [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ "$CLAUDE_CONFIG_DIR" != "${OSHIELD_CLAUDE_INJECTED_CONFIG_DIR:-}" ]; then
        command claude "$@"; return
      fi
      # Why: Shield's value replaced the user's own at spawn, so System default restores theirs.
      if [ -z "$__shield_claude_home" ] && [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ -n "${OSHIELD_CLAUDE_USER_CONFIG_DIR:-}" ]; then
        ( unset OSHIELD_CLAUDE_INJECTED_CONFIG_DIR; export CLAUDE_CONFIG_DIR="$OSHIELD_CLAUDE_USER_CONFIG_DIR"; command claude "$@" ); return
      fi
      if [ -z "$__shield_claude_home" ]; then
        ( unset CLAUDE_CONFIG_DIR OSHIELD_CLAUDE_INJECTED_CONFIG_DIR; command claude "$@" ); return
      fi
      case "$__shield_claude_home" in /*|[A-Za-z]:*) ;; *) return 1 ;; esac
      # Why -m 700: matches the folder Shield's setup creates; a credentials folder stays private.
      [ -d "$__shield_claude_home" ] || mkdir -p -m 700 -- "$__shield_claude_home" 2>/dev/null
      ( export CLAUDE_CONFIG_DIR="$__shield_claude_home" OSHIELD_CLAUDE_INJECTED_CONFIG_DIR="$__shield_claude_home"; command claude "$@" )
    }
  fi
  unset __shield_claude_binary
  if [[ -n "${_shield_histfile:-}" ]]; then
    HISTFILE="$_shield_histfile"
  fi
  if __shield_has_feature ready; then
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__shield_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __shield_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __shield_prev_line_init_fn=""
    fi
    __shield_prompt_mark() {
      printf "\033]777;shield-shell-ready\007"
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__shield_prev_line_init_fn:-}" ]]; then
        "${__shield_prev_line_init_fn}" "$@"
      fi
    }
    zle -N zle-line-init __shield_prompt_mark
  fi
  builtin unset _shield_shell_features _shield_histfile
  (( $+__shield_deferred_sched_armed )) || builtin unfunction __shield_deferred_sched_init
  builtin unfunction __shield_deferred_init __shield_has_feature __shield_arm_deferred_line_init
}
{
  builtin typeset _shield_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_shield_user_zshenv" ]] || builtin source -- "$_shield_user_zshenv"
} always {
  builtin unset _shield_user_zshenv
  if (( ! $+_shield_deferred_init_done )); then
    builtin typeset -ag precmd_functions
    (( ${precmd_functions[(Ie)__shield_deferred_init]} )) || precmd_functions+=(__shield_deferred_init)
    __shield_arm_deferred_line_init
  fi
}
